ソース 1: キャラクターを右方向に移動距離 100 (ピクセル/秒)で動かす 
extends Actor

# 移動速度 (ピクセル/秒)
# export をつけるとTAROのインスペクタで数値を変更できるようになる
@export var speed = 100 

# onready をつけると開始時に表示領域のサイズを取得する
@onready var screen_size = get_viewport_rect().size 

# 各フレームごとに繰り返し実行される関数
func _process(delta):
	# 移動方向・速度をベクトルで指定する
	# 方向をnormalized 関数で正規化してからspeedをかけると指定した方向に speed(ピクセル/秒)の速さで動く
	var velocity = Vector2(1,0).normalized()*speed 
	# 位置を更新する
	# deltaは前フレームからの経過時間(秒)
	# deltaを速さにかけることでFPSの影響を受けずに移動距離を一定にできる(距離=速さx時間)
	position += velocity*delta
	# 画面外に出ないようにクランプする
	position = position.clamp(Vector2.ZERO, screen_size)
